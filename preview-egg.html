// TENSOR - bare metal
pub struct Tensor {
    data: Vec<f32>,
    shape: Vec<usize>,
}

impl Tensor {
    pub fn zeros(shape: &[usize]) -> Self {
        let size = shape.iter().product();
        Tensor {
            data: vec![0.0; size],
            shape: shape.to_vec(),
        }
    }

    pub fn randn(shape: &[usize]) -> Self {
        let size = shape.iter().product();
        let mut t = Tensor::zeros(shape);
        let mut seed: u64 = 42;
        for i in 0..size {
            seed = seed.wrapping_mul(6364136223846793005)
                      .wrapping_add(1);
            let u = (seed >> 33) as f32 / 4294967296.0;
            t.data[i] = (u - 0.5) * 2.0;
        }
        t
    }

    pub fn matmul(&self, other: &Tensor) -> Tensor {
        let (m, k1) = (self.shape[0], self.shape[1]);
        let (k2, n) = (other.shape[0], other.shape[1]);
        assert_eq!(k1, k2);
        let mut out = Tensor::zeros(&[m, n]);
        for i in 0..m {
            for j in 0..n {
                let mut sum = 0.0;
                for k in 0..k1 {
                    sum += self.data[i*k1+k]
                         * other.data[k*n+j];
                }
                out.data[i*n+j] = sum;
            }
        }
        out
    }
}

// DENSE LAYER
pub struct Dense {
    weights: Tensor,
    bias: Tensor,
}

impl Dense {
    pub fn forward(&self, x: &Tensor) -> Tensor {
        let mut out = x.matmul(&self.weights);
        for i in 0..out.data.len() {
            out.data[i] = relu(out.data[i]);
        }
        out
    }
}
